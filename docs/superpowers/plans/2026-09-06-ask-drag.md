# Drag-to-place questions (`ask` + `widget: "drag"`): Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A question where the viewer drags names from a list onto the figure — organs onto the body, country names onto the map, note names onto the keyboard — judged per drop (in place / close / off) and as a whole, with the truth revealed in two-colour glow.

**Architecture:** A pure model (`drag-model.ts`) resolves each item to a target — an element's outline and box from the layout, or a piano key / chess square from the widget helpers — and judges a drop point against it. A DOM gate (`drag-gate.ts`) is the fifth answer device beside click, piano, chess and code: chips in a tray over the stage, pointer drag with capture, a verdict per chip, then it resolves with the hit ids. The planner makes the ask step carry the items and their boxes and implies `answer`; the player shows the element items, glows hits green and misses red while the answer line is spoken, and taps each box in movies.

**Tech Stack:** TypeScript, Vite, Vitest (node environment — DOM pieces are checked in the browser smoke, pure pieces in tests).

**Spec:** `docs/superpowers/specs/2026-09-06-ask-drag-design.md`

```ts
// src/ui/drag-model.ts
// The drag question's rules, DOM-free: what each chip's target is (an element's
// outline and box, or a piano key / chess square by name), how a drop point is
// judged against it, and the words the card uses. The gate (drag-gate.ts)
// and the planner both lean on this; node tests cover it.

import type { BBox } from "../layout/geometry";
import type { Pt } from "../layout/model";
import { pointInRing } from "./hit";

export interface DragItem { id: string; label: string }
export interface DragTarget { id: string; label: string; box: BBox; rings?: Pt[][]; /** An element of the figure (can be shown and glowed) — false for a note or a square. */ element: boolean }
export type DragGrade = "in" | "near" | "far";
export interface DragJudgement { hit: boolean; grade: DragGrade; /** Distance outside the outline as a fraction of the target's bbox diagonal; 0 inside. */ distance: number }

/** "kidney_left" → "Kidney left". Authors give better labels; this is the fallback. */
export function humanLabel(id: string): string {
  const s = id.replace(/_+/g, " ").trim();
  return s.charAt(0).toUpperCase() + s.slice(1);
}

export function normalizeItems(items: (string | { id: string; label?: string })[]): DragItem[] {
  return items.map((it) => (typeof it === "string" ? { id: it, label: humanLabel(it) } : { id: it.id, label: it.label ?? humanLabel(it.id) }));
}

/** Element ids first (the layout's boxes; outlines when it has them), then a
 *  note on a piano figure, then a square on a chess figure. */
export function resolveDragTargets(
  items: DragItem[],
  geo: { boxes: ReadonlyMap<string, BBox>; rings: ReadonlyMap<string, Pt[][]>; noteBox?: (note: string) => BBox | null; squareBox?: (square: string) => BBox | null },
): { targets: DragTarget[]; missing: string[] } {
  const targets: DragTarget[] = [];
  const missing: string[] = [];
  for (const it of items) {
    const box = geo.boxes.get(it.id);
    if (box) {
      const rings = geo.rings.get(it.id);
      targets.push({ id: it.id, label: it.label, box, ...(rings ? { rings } : {}), element: true });
      continue;
    }
    const other = geo.noteBox?.(it.id) ?? geo.squareBox?.(it.id) ?? null;
    if (other) targets.push({ id: it.id, label: it.label, box: other, element: false });
    else missing.push(it.id);
  }
  return { targets, missing };
}

const distToSegment = (p: Pt, a: Pt, b: Pt): number => {
  const vx = b[0] - a[0], vy = b[1] - a[1];
  const len2 = vx * vx + vy * vy;
  const t = len2 === 0 ? 0 : Math.max(0, Math.min(1, ((p[0] - a[0]) * vx + (p[1] - a[1]) * vy) / len2));
  return Math.hypot(p[0] - (a[0] + t * vx), p[1] - (a[1] + t * vy));
};
const distToRing = (p: Pt, ring: Pt[]): number => {
  let d = Infinity;
  for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) d = Math.min(d, distToSegment(p, ring[j], ring[i]));
  return d;
};
const inBox = (p: Pt, b: BBox) => p[0] >= b.x && p[0] <= b.x + b.w && p[1] >= b.y && p[1] <= b.y + b.h;
const distToBox = (p: Pt, b: BBox) => Math.hypot(Math.max(b.x - p[0], 0, p[0] - (b.x + b.w)), Math.max(b.y - p[1], 0, p[1] - (b.y + b.h)));

/** Inside (even-odd over the rings, or the box) → in. Else the distance to the
 *  outline over the bbox diagonal: within tolerance → near (a hit), else far. */
export function judgeDrop(p: Pt, target: Pick<DragTarget, "box" | "rings">, tolerance: number): DragJudgement {
  const rings = target.rings?.filter((r) => r.length >= 3);
  const inside = rings && rings.length > 0 ? rings.reduce((acc, r) => acc !== pointInRing(r, p), false) : inBox(p, target.box);
  if (inside) return { hit: true, grade: "in", distance: 0 };
  const d = rings && rings.length > 0 ? Math.min(...rings.map((r) => distToRing(p, r))) : distToBox(p, target.box);
  const diag = Math.hypot(target.box.w, target.box.h) || 1;
  const distance = d / diag;
  const hit = distance <= tolerance;
  return { hit, grade: hit ? "near" : "far", distance };
}

export function dragSummary(judged: DragJudgement[]): string {
  const hits = judged.filter((j) => j.hit).length;
  const n = judged.length;
  if (n === 1) return hits === 1 ? "In place" : "Not quite";
  return hits === n ? `All ${n} in place` : `${hits} of ${n} in place`;
}
```

```ts
// src/ui/drag-gate.ts
// The drag question's answer device: a tray of chips over the figure's left
// edge, each dragged (pointer capture, touch included) onto the figure and
// judged where it lands — in place, close, or off — then left there with its
// verdict. When every chip has landed the card sums up, lingers, and resolves
// with the hit ids. The rules are drag-model.ts; this file is only the DOM.

import type { RenderHandle } from "../render";
import { elementBBoxes, elementRings } from "../layout/layout";
import { makeBrowserMeasure } from "../render/svg-backend";
import { chessSquareBox, pianoKeyBox, pianoOctaves } from "../render/widgets";
import { h, logicalPoint } from "./dom";
import { dragSummary, judgeDrop, resolveDragTargets, type DragJudgement, type DragTarget } from "./drag-model";
import type { AskGateStep } from "./controls";

const LINGER_MS = 2600; // matches the other cards' CARD_LINGER_MS
const GRADE_WORD: Record<DragJudgement["grade"], string> = { in: "in place", near: "close", far: "off" };

export function dragGateFor(stage: HTMLElement, hd: RenderHandle): (signal: AbortSignal, step: AskGateStep) => Promise<string | null> {
  return (signal, step) =>
    new Promise<string | null>((resolve) => {
      stage.querySelector(".cs-figgate")?.remove();
      const items = step.items ?? [];
      const { targets } = resolveDragTargets(items, {
        boxes: elementBBoxes(hd.layout, makeBrowserMeasure()),
        rings: elementRings(hd.layout),
        noteBox: (n) => pianoKeyBox(pianoOctaves(hd.spec.params), n),
        squareBox: (s) => chessSquareBox(hd.spec.params?.["flip"] === true, s),
      });
      const tolerance = step.tolerance ?? 0.25;
      const hint = h("span", { class: "cs-waitgate-pill cs-figgate-hint" }, "Drag each name onto the figure ▸");
      const summary = h("span", { class: "cs-waitgate-pill cs-drag-summary", hidden: "" });
      const tray = h("div", { class: "cs-drag-tray" });
      const gate = h("div", { class: "cs-figgate cs-draggate" }, tray, hint, summary);
      let settled = false;
      const remove = (): void => {
        signal.removeEventListener("abort", onAbort);
        gate.remove();
      };
      const onAbort = (): void => {
        remove();
        if (!settled) {
          settled = true;
          resolve(null);
        }
      };
      const judged = new Map<string, DragJudgement>();
      const finish = (): void => {
        if (settled) return;
        settled = true;
        hint.remove();
        summary.textContent = dragSummary(targets.map((t) => judged.get(t.id)!));
        summary.hidden = false;
        window.setTimeout(remove, LINGER_MS);
        resolve(targets.filter((t) => judged.get(t.id)!.hit).map((t) => t.id).join(","));
      };
      if (targets.length === 0) {
        // Nothing to place (every item was skipped by the planner): no gate, no answer.
        settled = true;
        resolve(null);
        return;
      }
      for (const t of targets) tray.appendChild(chipFor(t));
      if (!step.required) {
        const skip = h("button", { class: "cs-cardgate-pill skip cs-figgate-skip" }, "Skip ▸");
        skip.addEventListener("click", (e) => {
          e.stopPropagation();
          if (settled) return;
          settled = true;
          remove();
          resolve(null);
        });
        gate.appendChild(skip);
      }
      gate.addEventListener("click", (e) => e.stopPropagation());
      signal.addEventListener("abort", onAbort);
      stage.appendChild(gate);

      function chipFor(t: DragTarget): HTMLElement {
        const chip = h("button", { class: "cs-cardgate-pill cs-drag-chip", type: "button", "data-id": t.id }, t.label);
        let start: [number, number] | null = null;
        let dx = 0, dy = 0;
        chip.addEventListener("pointerdown", (e) => {
          if (settled || judged.has(t.id)) return;
          e.preventDefault();
          e.stopPropagation();
          chip.setPointerCapture(e.pointerId);
          start = [e.clientX - dx, e.clientY - dy];
          chip.classList.add("dragging");
        });
        chip.addEventListener("pointermove", (e) => {
          if (!start) return;
          dx = e.clientX - start[0];
          dy = e.clientY - start[1];
          chip.style.transform = `translate(${dx}px, ${dy}px)`;
        });
        const drop = (e: PointerEvent): void => {
          if (!start) return;
          start = null;
          chip.classList.remove("dragging");
          const p = logicalPoint(stage, e);
          const r = stage.getBoundingClientRect();
          const onStage = e.clientX >= r.left && e.clientX <= r.right && e.clientY >= r.top && e.clientY <= r.bottom;
          if (!p || !onStage || (Math.abs(dx) < 4 && Math.abs(dy) < 4)) {
            // Off the figure, or a mere click: back to the tray.
            dx = 0; dy = 0;
            chip.style.transform = "";
            return;
          }
          const j = judgeDrop(p, t, tolerance);
          judged.set(t.id, j);
          chip.classList.add("placed", j.hit ? "right" : "wrong");
          chip.title = GRADE_WORD[j.grade];
          chip.append(h("span", { class: "cs-drag-verdict" }, j.hit ? (j.grade === "in" ? " ✓" : " ✓ close") : " ✗"));
          if (judged.size === targets.length) finish();
        };
        chip.addEventListener("pointerup", drop);
        chip.addEventListener("pointercancel", () => {
          start = null;
          chip.classList.remove("dragging");
          dx = 0; dy = 0;
          chip.style.transform = "";
        });
        return chip;
      }
    });
}
```

Notes: `h()`'s attribute `hidden: ""` — check how `h` sets boolean attributes (the codebase uses `el.hidden = true` elsewhere); if `hidden: ""` is not supported, set `summary.hidden = true` after creation instead. The chip keeps its `transform` after the drop, so it stays where it landed; the tray is `position: absolute` in the gate, chips are in normal flow inside it.
